type Fixture = Awaited<ReturnType<typeof fixture>>

function firstEntry(stored: SfProductState) {
  const entries = Object.values(stored.outbox)
  expect(entries).toHaveLength(1)
  const entry = entries[0]!
  expect(entry.choiceCapture).toBeDefined()
  if (!entry.choiceCapture) throw new Error("Expected the actual protected choice capture")
  return { entry, capture: entry.choiceCapture }
}
async function recorded(f: Fixture) {
  expect((await f.record()).status).toBe("ok")
  const stored = await f.readStored()
  expect(stored.cases[originalCaseRef.id]!.choices).toHaveLength(1)
  expect(stored.cases[originalCaseRef.id]!.version).toBe(5)
  return { stored, ...firstEntry(stored) }
}
function rebindSelection(selection: CaptureSelection) {
  selection.event.evidenceRefs = structuredClone(selection.choice.evidenceRefs)
  selection.event.outputDigest = digestSfValue(selection.choice)
  selection.case.choices = [structuredClone(selection.choice)]
  selection.case.ledger = [structuredClone(selection.event)]
}

test("one canonical choice atomically captures original Planner evidence in the cases/outbox commit", async () => {
  const f = await fixture()
  try {
    let returned: readonly SfLedgerRecordCapture[] | undefined
    f.controls.captureHook = (_selection, result) => { if (result.status === "ok") returned = result.value }
    const { stored, entry, capture } = await recorded(f)
    const decision = stored.cases[originalCaseRef.id]!
    const choice = decision.choices[0]!, event = decision.ledger[0]!
    expect(f.controls.captureCalls).toBe(1)
    expect(entry.state).toBe("pending")
    expect(capture.records).toEqual(f.captures)
    expect(capture.choiceId).toBe(choice.id)
    expect(capture.outputDigest).toBe(digestSfValue(choice))
    expect(capture.eventRefDigest).toBe(entry.eventRef.refDigest)
    expect(capture.eventDigest).toBe(digestSfValue(event))
    expect(event.outputDigest).toBe(capture.outputDigest)
    expect(capture.captureDigest).toBe(sfChoiceSourceCaptureDigest(entry, capture))
    expect(stored.versions.cases).toBe(2)
    expect(stored.versions.outbox).toBe(1)
    expect(decision.preparedChanges).toEqual([])
    expect(decision.receipts).toEqual([])
    expect(stored.atlas).toEqual({})
    returned![0]!.projection.values[0]!.value = 999999
    returned![0]!.ref.target.version += 50
    expect((await f.readStored()).outbox).toEqual(stored.outbox)
    const read = await f.readProtected(f.refs[1]!.refDigest, f.context)
    expect(read).toEqual(f.captures[1])
    read!.projection.values[0]!.value = -1
    expect(await f.readProtected(f.refs[1]!.refDigest, f.context)).toEqual(f.captures[1])
  } finally { await f.close() }
})

test("exact retry keeps the original choice and capture and never captures again", async () => {
  const f = await fixture()
  try {
    const original = await recorded(f)
    const calls = f.controls.captureCalls
    f.mutationInput.captureChoiceSources = async () => { throw new Error("Replay must not enrich historical evidence") }
    expect((await f.record()).status).toBe("ok")
    expect(f.controls.captureCalls).toBe(calls)
    expect(await f.readStored()).toEqual(original.stored)
  } finally { await f.close() }
})

test("legacy choice entries remain uncaptured on retry and cannot synthesize an original projection", async () => {
  const f = await fixture()
  try {
    const realCapture = f.mutationInput.captureChoiceSources
    delete f.mutationInput.captureChoiceSources
    expect((await f.record()).status).toBe("ok")
    const before = await f.readStored()
    expect(Object.values(before.outbox)).toHaveLength(1)
    expect(Object.values(before.outbox)[0]!.choiceCapture).toBeUndefined()
    f.mutationInput.captureChoiceSources = realCapture
    expect((await f.record()).status).toBe("ok")
    expect(await f.readStored()).toEqual(before)
    expect(await f.readProtected(f.refs[1]!.refDigest, f.context)).toBeUndefined()
    expect((await f.history.resolve(f.refs[1]!, f.context)).state).toBe("unavailable")
  } finally { await f.close() }
})

test.each(["source-denied", "incomplete-records", "not-an-array", "selection-mutation", "headers-drift", "lease-ended", "capture-method-replaced"] as const)(
  "capture failure %s commits neither a choice nor any outbox effect", async (failure) => {
    const f = await fixture()
    try {
      const before = await f.readStored()
      if (failure === "source-denied") f.controls.projectorDenied = true
      f.controls.captureHook = (selection, result) => {
        if (failure === "incomplete-records" && result.status === "ok") result.value = result.value.slice(0, 1)
        if (failure === "not-an-array" && result.status === "ok") Object.assign(result, { value: undefined })
        if (failure === "selection-mutation") selection.choice.reason = "Changed after proof"
        if (failure === "headers-drift") f.transport.headers.set("x-unit-authority", "changed")
        if (failure === "lease-ended") f.ui.finish(f.context)
        if (failure === "capture-method-replaced") f.mutationInput.captureChoiceSources = async () => ({ status: "ok", value: [] })
      }
      expect((await f.record()).status).not.toBe("ok")
      expect(await f.readStored()).toEqual(before)
    } finally { await f.close() }
  },
)

test.each(["source-argument", "projector-method", "headers"] as const)("capture refuses %s drift while awaiting the authorized projector", async (failure) => {
  const f = await fixture()
  try {

function gate() {
  let release!: () => void
  const promise = new Promise<void>((resolve) => { release = resolve })
  return { promise, release }
}

test.each(["choice", "outbox-entry", "captured-source"] as const)(
  "final capture egress serializes a concurrent change to %s before exposing original bytes", async (target) => {
    const f = await fixture()
    const start = gate(), started = gate()
    let triggered = false, held = false, writerEntered = false, writerCommitted = false
    let writer: Promise<void> | undefined
    try {
      await recorded(f)
      f.controls.readNamespaces = []
      f.controls.completedCaseAuthorizations = 0
      f.controls.monitorReads = true
      // This promise and its first await are created OUTSIDE every C08 read.
      // The competing writer therefore acquires its own canonical transaction;
      // it is not a reentrant write hidden inside the reader's snapshot.
      writer = (async () => {
        await start.promise
        started.release()
        expect(f.sources.currentSnapshots?.()).toBeUndefined()
        await f.corrupt((stored) => {
          writerEntered = true
          if (target === "choice") stored.cases[originalCaseRef.id]!.choices[0]!.reason = "Concurrent human choice revision"
          if (target === "outbox-entry") delete stored.outbox[Object.keys(stored.outbox)[0]!]
          if (target === "captured-source") Object.values(stored.outbox)[0]!.choiceCapture!.records[1]!.projection.values[0]!.value = -77
        })
        writerCommitted = true
      })()
      // Attach a rejection observer immediately; the original promise is still
      // awaited below, so a failed writer assertion cannot disappear.
      void writer.catch(() => undefined)
      f.controls.revalidateHook = async () => {
        if (triggered || f.controls.readNamespaces.filter((name) => name === "outbox").length < 2 ||
          f.controls.completedCaseAuthorizations < 2) return
        triggered = true
        held = f.sources.currentSnapshots?.() !== undefined
        start.release()
        await started.promise
        if (held) {
          expect(writerEntered).toBe(false)
          expect(writerCommitted).toBe(false)
        } else {
          // Reproduce the former post-read await: mutation is now observable
          // before the revalidation returns. Such an egress must refuse.
          await writer
        }
      }
      const result = await f.readProtected(f.refs[1]!.refDigest, f.context)
      expect(triggered).toBe(true)
      expect(result === undefined || held).toBe(true)
      if (result !== undefined) expect(result).toEqual(f.captures[1])
      f.controls.revalidateHook = undefined
      f.controls.monitorReads = false
      start.release()
      await writer
      expect(writerEntered).toBe(true)
      expect(writerCommitted).toBe(true)
      expect(await f.readProtected(f.refs[1]!.refDigest, f.context)).toBeUndefined()
    } finally {
      f.controls.revalidateHook = undefined
      start.release()
      await writer?.catch(() => undefined)
      await f.close()
    }
  },
)
