import { AsyncLocalStorage } from "node:async_hooks"

// SF52 diagnostic16 only: aggregate fixed stage names, counts and durations.
// No payload, identifier, credential, projection or authority value is retained.
type Stage = { count: number; total_ms: number; max_ms: number }
const profiles = new AsyncLocalStorage<Record<string, Stage>>()

export function sf52Measure<T>(stage: string, work: () => T): T {
  const stats = profiles.getStore()
  if (!stats) return work()
  const at = performance.now()
  try {
    return work()
  } finally {
    const elapsed = performance.now() - at
    const row = stats[stage] ??= { count: 0, total_ms: 0, max_ms: 0 }
    row.count += 1
    row.total_ms += elapsed
    row.max_ms = Math.max(row.max_ms, elapsed)
  }
}

export async function sf52Profile<T>(work: () => Promise<T>): Promise<T> {
  const stats: Record<string, Stage> = Object.create(null)
  const at = performance.now()
  try {
    return await profiles.run(stats, work)
  } finally {
    // An unavailable diagnostic sink cannot replace the producer's return/throw.
    try { console.info("SF52_COST", JSON.stringify({ diagnostic: 16, elapsed_ms: performance.now() - at, stages: stats })) } catch { /* Timing only. */ }
  }
}
